# Sandbox parser dependencies — pinned graph for Dockerfile.sandbox.
#
# Audit R3-19 — the prior Dockerfile.sandbox inlined `pip install pandas
# pyarrow openpyxl` which left every transitive (numpy, dateutil, pytz,
# etc.) unpinned and invisible to trivy/pip-audit. With this lockfile the
# sandbox image's full dependency tree is in git, scanned at CI time, and
# reproducible across builds.
#
# Versions pinned to match the main pipeline (requirements.txt) so a
# parquet round-trip from training → sandbox can never hit binary-format
# drift (closes R3-3 at the version-graph layer, not just at the pyarrow
# leaf).
#
# Regenerate via:
#   docker run --rm -v $(pwd)/docker:/out python:3.11-slim sh -c \
#     "pip install --no-cache-dir pandas==2.2.2 pyarrow==15.0.2 openpyxl==3.1.2 && \
#      pip freeze > /out/requirements-sandbox.txt"
# ═══════════════════════════════════════════════════════════════════════════

# ── Direct deps ──────────────────────────────────────────────────────────
pandas==2.2.2
pyarrow==15.0.2
openpyxl==3.1.2

# ── Transitives (pinned for reproducibility + scan visibility) ──────────
# Versions match the resolved graph from pandas==2.2.2 + pyarrow==15.0.2
# on python:3.11-slim 2026-05. Bump in lockstep with the main
# requirements.txt — sandbox + training pipeline read each other's
# parquet, so numpy/pyarrow drift breaks deserialization silently.
numpy==1.26.4
python-dateutil==2.9.0.post0
pytz==2024.1
tzdata==2024.1
et-xmlfile==1.1.0
